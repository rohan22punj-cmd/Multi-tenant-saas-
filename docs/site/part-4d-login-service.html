    Part 4d: Login Service & Timing Attack Protection (src/services/authService.js)

    src/services/authService.js

    The login() function handles user authentication. It prevents user and company enumeration via timing attacks and ensures all failures return the same generic error message.

    const DUMMY_HASH = '$2a$12$dummyhashdummyhashdummyh';

export async function login({ companySlug, email, password }) {
  // 1. Find the tenant (unscoped - Tenant model has no tenantId)
  const tenant = await runUnscoped(
    'login: find tenant by slug',
    () => Tenant.findOne({ slug: companySlug })
  );

  // 2. Find the user within that tenant INSIDE tenant context
  let user = null;
  if (tenant) {
    user = await runWithTenant({ tenantId: tenant._id }, async () => {
      // tenantId is auto-injected by tenantPlugin; we don't pass it
      return User.findOne({ email });
    });
  }

  // 3. Always run bcrypt.compare to prevent timing attacks.
  // If user doesn't exist or is inactive, compare against a dummy hash.
  const hashToCompare = (user && user.isActive) ? user.passwordHash : DUMMY_HASH;
  const passwordMatch = await bcrypt.compare(password, hashToCompare);

  // 4. If any check failed, return the SAME generic message.
  // This prevents attackers from discovering which companies/emails exist.
  if (!tenant || !user || !user.isActive || !passwordMatch) {
    throw new AppError('Invalid credentials', 401);
  }

  // 5. Fetch the role to get current permissions (inside tenant context)
  const role = await runWithTenant({ tenantId: tenant._id }, async () => {
    return Role.findById(user.roleId).lean();
  });
  if (!role) {
    throw new AppError('Invalid credentials', 401);
  }

  // 6. Issue tokens and store refresh token hash (inside tenant context)
  const tokens = await runWithTenant({ tenantId: tenant._id }, async () => {
    return issueAndStoreTokens(user, role.permissions);
  });

  // 7. Return tokens + minimal user object (no password hash)
  return {
    ...tokens,
    user: {
      id: user._id,
      name: user.name,
      email: user.email,
      role: role.name,
    },
  };
}

    Explanation:

    • Dummy bcrypt compare (DUMMY_HASH): If a user does not exist or is inactive, the function still runs bcrypt.compare against a dummy hash. This ensures the function takes the same amount of time whether the user exists or not, preventing timing attacks that could reveal valid usernames or companies.
    • Company lookup (unscoped): The tenant is looked up using the provided companySlug via runUnscoped, as the Tenant model is not tenant-scoped (it's the top level).
    • User lookup inside tenant context: The user is looked up inside the tenant context using runWithTenant, ensuring the query is scoped to the correct tenant.
    • Generic error message: All failures (invalid tenant, user, inactive user, wrong password) return the same "Invalid credentials" error to prevent attackers from distinguishing between different failure types.
