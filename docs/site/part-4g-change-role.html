    Part 4g: Change Role & Deactivate (userController.js)

    What this covers: The deactivate, reactivate, and changeRole handlers in src/controllers/userController.js. These endpoints manage user status and permissions with strict safety rules to prevent lockouts and privilege escalation.

    Overview

    Three endpoints exist under PATCH /api/users/:id:

    • /deactivate — sets isActive = false and wipes refreshTokenHash
    • /reactivate — sets isActive = true (idempotent)
    • /role — changes a user's roleId and wipes refreshTokenHash

    Per-Tenant Mutex: withTenantLock

    Because the "last admin" checks are read-then-write operations, two concurrent requests could both pass the check and leave the tenant with zero admins. withTenantLock serializes these operations per tenant using an in-process promise chain.

    // src/controllers/userController.js (lines 44-71)

const tenantLocks = new Map();

function withTenantLock(tenantId, task) {
  const key = String(tenantId);
  const previous = tenantLocks.get(key) || Promise.resolve();
  const run = previous.then(() => task());

  // Keep the chain alive whether the task fulfilled or rejected
  const tail = run.catch(() => {});
  tenantLocks.set(key, tail);
  tail.then(() => {
    if (tenantLocks.get(key) === tail) tenantLocks.delete(key);
  });

  return run;
}
    Limitation: This only serializes requests within a single Node process. In a multi-instance deployment, replace with a distributed lock (Redis SET NX or MongoDB lease). The call sites stay the same.

    Deactivate — PATCH /api/users/:id/deactivate

    Source

    // src/controllers/userController.js (lines 79-129)

export const deactivate = asyncHandler(async (req, res) => {
  const targetId = req.params.id;
  const callerId = req.user.userId;

  // Rule a: cannot deactivate yourself
  if (targetId === callerId) {
    throw new AppError('Forbidden', 403);
  }

  // tenantPlugin scopes this to the caller's tenant — another company's id → null → 404
  const target = await User.findById(targetId);
  if (!target) {
    throw new AppError('User not found', 404);
  }

  // Rule c: privilege check — caller must have every permission the target's role has
  const targetRole = await Role.findById(target.roleId).lean();
  if (targetRole) {
    const callerPerms = new Set(req.user.permissions);
    const hasPrivilegeGap = targetRole.permissions.some((p) => !callerPerms.has(p));
    if (hasPrivilegeGap) {
      throw new AppError('Forbidden', 403);
    }
  }

  // Rule b: cannot deactivate the last active user who holds USER_MANAGE
  if (target.isActive) {
    const adminRoles = await Role.find({ permissions: PERMISSIONS.USER_MANAGE }).lean();
    const adminRoleIds = adminRoles.map((r) => r._id.toString());

    const remainingAdminCount = await User.countDocuments({
      _id: { $ne: target._id },
      isActive: true,
      roleId: { $in: adminRoleIds },
    });

    if (remainingAdminCount === 0 && adminRoleIds.includes(target.roleId.toString())) {
      throw new AppError('Cannot deactivate the last admin', 409);
    }
  }

  // All checks passed — deactivate and cut off future refresh token use
  target.isActive = false;
  target.refreshTokenHash = null;
  await target.save();

  res.status(200).json({ id: target._id, isActive: target.isActive });
});

    Reactivate — PATCH /api/users/:id/reactivate

export const reactivate = asyncHandler(async (req, res) => {

    throw new AppError('User not found', 404);

    Change Role — PATCH /api/users/:id/role

    Source

    // src/controllers/userController.js (lines 168-256)

export const changeRole = asyncHandler(async (req, res) => {
  const targetId = req.params.id;
  const { roleId } = req.body;
  const callerId = req.user.userId;

  // Validate roleId is a well-formed ObjectId
  if (!mongoose.Types.ObjectId.isValid(roleId)) {
    throw new AppError('Invalid role id', 400);
  }

  // Rule a: cannot change your own role
  if (targetId === callerId) {
    throw new AppError('Forbidden', 403);
  }

  // Serialize the check + write per tenant
  return withTenantLock(req.user.tenantId, async () => {
    const target = await User.findById(targetId).populate('roleId');
    if (!target) {
      throw new AppError('User not found', 404);
    }

    const newRole = await Role.findById(roleId).lean();
    if (!newRole) {
      throw new AppError('Role not found', 404);
    }

    // Rule e: already has that role → idempotent success
    if (target.roleId._id.toString() === roleId) {
      return res.status(200).json({
        id: target._id,
        roleId: target.roleId._id,
        roleName: target.roleId.name,
      });
    }

    const callerPerms = new Set(req.user.permissions);

    // Rule b: privilege check on NEW role — cannot assign permissions you don't have
    const hasEscalationInNewRole = newRole.permissions.some((p) => !callerPerms.has(p));
    if (hasEscalationInNewRole) {
      throw new AppError('Forbidden', 403);
    }

    // Rule c: privilege check on CURRENT role — cannot change someone with more permissions
    const currentRole = target.roleId;
    const hasPrivilegeGapInCurrentRole = currentRole.permissions.some((p) => !callerPerms.has(p));
    if (hasPrivilegeGapInCurrentRole) {
      throw new AppError('Forbidden', 403);
    }

    // Rule d: last-admin check — only if target IS an admin and we're demoting
    const targetCurrentlyHasUserManage = currentRole.permissions.includes(PERMISSIONS.USER_MANAGE);
    const newRoleHasUserManage = newRole.permissions.includes(PERMISSIONS.USER_MANAGE);

    if (targetCurrentlyHasUserManage && !newRoleHasUserManage && target.isActive) {
      const adminRoles = await Role.find({ permissions: PERMISSIONS.USER_MANAGE }).lean();
      const adminRoleIds = adminRoles.map((r) => r._id.toString());

      const remainingAdminCount = await User.countDocuments({
        _id: { $ne: target._id },
        isActive: true,
        roleId: { $in: adminRoleIds },
      });

      if (remainingAdminCount === 0) {
        throw new AppError('Cannot demote the last admin', 409);
      }
    }

    // All checks passed — update role and wipe refresh token
    target.roleId = roleId;
    target.refreshTokenHash = null; // forces re-login to get new permissions
    await target.save();

    // Re-fetch to get the new role's name for the response
    const updatedUser = await User.findById(targetId).populate('roleId');

    res.status(200).json({
      id: updatedUser._id,
      roleId: updatedUser.roleId._id,
      roleName: updatedUser.roleId.name,
    });
  });
});

    Refresh Token Wipe

    Both deactivate and changeRole set refreshTokenHash = null. This means:

    • The user's access token remains valid for up to 15 minutes (stateless JWT, no DB check on every request)
    • The user cannot obtain a new access token after that — the refresh token is dead
    • This is a deliberate trade-off: near-instant cutoff would require a Redis blocklist checked on every request
    // src/controllers/userController.js (lines 124-126, 243-245)

target.isActive = false;
target.refreshTokenHash = null;
await target.save();

// ...

target.roleId = roleId;

await target.save();
    Ponytail note (from source): Access tokens are still valid up to 15 min post-deactivation; add a token revocation list (e.g. Redis blocklist) if near-instant cutoff is required.
