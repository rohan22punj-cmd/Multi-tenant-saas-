    1. Tenant Isolation — tests/tenantIsolation.test.js

    Unit Level (Direct Model Calls)

    Uses runWithTenant to set context and verifies the tenant plugin scopes every operation:

    // tests/tenantIsolation.test.js (excerpts)

describe('Tenant Isolation — Unit Level', () => {
  let tenantA, tenantB, roleA, roleB, userA1, userA2, userB1;

  beforeEach(async () => {
    tenantA = await createTenant('Company A', 'company-a');
    tenantB = await createTenant('Company B', 'company-b');
    roleA = await createRoleInContext(tenantA._id, 'admin', ALL_PERMISSIONS);
    roleB = await createRoleInContext(tenantB._id, 'admin', ALL_PERMISSIONS);
    userA1 = await createUserInContext(tenantA._id, 'Alice A1', 'alice@a.com', PASSWORD_HASH, roleA._id);
    userA2 = await createUserInContext(tenantA._id, 'Alice A2', 'alice2@a.com', PASSWORD_HASH, roleA._id);
    userB1 = await createUserInContext(tenantB._id, 'Bob B1', 'bob@b.com', PASSWORD_HASH, roleB._id);
  });

  it('inside A context, find() returns only A users', async () => {
    const users = await runWithTenant({ tenantId: tenantA._id }, async () => User.find().lean());
    expect(users).toHaveLength(2);
    expect(users.every(u => u.tenantId.equals(tenantA._id))).toBe(true);
  });

  it('inside A context, findById with B user id returns null', async () => {
    const user = await runWithTenant({ tenantId: tenantA._id }, async () => User.findById(userB1._id).lean());
    expect(user).toBeNull();
  });

  it('inside A context, updateOne on B user matches 0 and does not modify', async () => {
    const result = await runWithTenant({ tenantId: tenantA._id }, async () =>
      User.updateOne({ _id: userB1._id }, { name: 'HACKED' })
    );
    expect(result.matchedCount).toBe(0);
    expect(result.modifiedCount).toBe(0);
  });

  it('inside A context, deleteOne on B user deletes 0', async () => {
    const result = await runWithTenant({ tenantId: tenantA._id }, async () =>
      User.deleteOne({ _id: userB1._id })
    );
    expect(result.deletedCount).toBe(0);
  });

  it('creating a document with explicit B tenantId in A context throws', async () => {
    await expect(
      runWithTenant({ tenantId: tenantA._id }, async () =>
        User.create({ ..., tenantId: tenantB._id })
      )
    ).rejects.toThrow(/does not match current context/i);
  });

  it('find() outside context throws', async () => {
    await expect(User.find().lean()).rejects.toThrow(/outside a tenant context/i);
  });
});

    Forced Overwrite Protection

    Even if a caller explicitly passes another tenant's ID, the plugin force-overwrites it to the current context:

    // tests/tenantIsolation.test.js (lines 111-117)

it('inside A context, findOne with explicit B tenantId filter returns null (forced overwrite)', async () => {
  const user = await runWithTenant({ tenantId: tenantA._id }, async () =>
    User.findOne({ tenantId: tenantB._id, email: 'bob@b.com' }).lean()
  );
  expect(user).toBeNull();
});

    HTTP Level — 20+ Concurrent Mixed Requests

    Logs in as Company A and Company B, fires 20 concurrent requests to /api/me, verifies zero data mixing:

    // tests/tenantIsolation.test.js (lines 459-509)

it('20+ mixed concurrent requests to /api/me never mix tenant data', async () => {
  const requests = [];

  for (let i = 0; i < 10; i++) {
    requests.push(request(app).get('/api/me').set('Authorization', `Bearer ${tokenA}`));
  }
  for (let i = 0; i < 10; i++) {
    requests.push(request(app).get('/api/me').set('Authorization', `Bearer ${tokenB}`));
  }

  const responses = await Promise.all(requests);

  // Verify all succeeded
  responses.forEach(res => expect(res.status).toBe(200));

  // Verify A's requests all see A's tenantId
  const aResponses = responses.slice(0, 10);
  aResponses.forEach(res => expect(res.body.data.tenantId).toBe(aTenantId));

  // Verify B's requests all see B's tenantId
  const bResponses = responses.slice(10, 20);
  bResponses.forEach(res => expect(res.body.data.tenantId).toBe(bTenantId));

  expect(aTenantId).not.toBe(bTenantId);
});

    2. Refresh & Logout — tests/refreshLogout.test.js

    Token Rotation

    // tests/refreshLogout.test.js (lines 57-67)

it('refreshes with a valid token → new access & refresh tokens', async () => {
  const { accessToken, refreshToken } = await signupAndLogin();
  const res = await request(app).post('/api/auth/refresh').send({ refreshToken });
  expect(res.status).toBe(200);
  expect(res.body.data.accessToken).toBeDefined();
  expect(res.body.data.refreshToken).toBeDefined();
  expect(res.body.data.refreshToken).not.toBe(refreshToken); // rotation
});

    Reuse Detection & Safety Wipe

    Using an old refresh token triggers a wipe — even the newest token stops working:

    // tests/refreshLogout.test.js (lines 81-109)

it('rejects reuse of the OLD refresh token with 401', async () => {
  const { refreshToken } = await signupAndLogin();
  const first = await request(app).post('/api/auth/refresh').send({ refreshToken });
  expect(first.status).toBe(200);
  const reuse = await request(app).post('/api/auth/refresh').send({ refreshToken });
  expect(reuse.status).toBe(401);
});

it('after reuse detection, even the NEWEST refresh token is dead (wipe)', async () => {
  const { refreshToken } = await signupAndLogin();
  const first = await request(app).post('/api/auth/refresh').send({ refreshToken });
  const newRefresh = first.body.data.refreshToken;

  await request(app).post('/api/auth/refresh').send({ refreshToken }); // triggers wipe

  const afterWipe = await request(app).post('/api/auth/refresh').send({ refreshToken: newRefresh });
  expect(afterWipe.status).toBe(401);
});

    Token Separation (Separate Secrets)

    // tests/refreshLogout.test.js (lines 156-171)

it('access token cannot be used as a refresh token', async () => {
  const { accessToken } = await signupAndLogin();
  const res = await request(app).post('/api/auth/refresh').send({ refreshToken: accessToken });
  expect(res.status).toBe(401);
});

it('refresh token cannot be used as an access token on a protected route', async () => {
  const { refreshToken } = await signupAndLogin();
  const res = await request(app).get('/api/me').set('Authorization', `Bearer ${refreshToken}`);
  expect(res.status).toBe(401);
});

    3. Plan Limit — tests/planLimit.test.js

    Free Plan Cap (Max 3 Projects)

    // tests/planLimit.test.js (lines 38-51)

test('the 4th project is rejected with 402 and the exact payload', async () => {
  for (const name of ['One', 'Two', 'Three']) {
    expect((await createProject(company.token, { name })).status).toBe(201);
  }
  const res = await createProject(company.token, { name: 'Four' });
  expect(res.status).toBe(402);
  expect(res.body).toEqual({
    error: 'Plan limit reached',
    limit: 3,
    plan: 'free',
  });

    Race Condition Test — Concurrent Creates

    Uses Promise.all to fire two simultaneous creates when the tenant is at 2/3 projects. Only one should succeed:

    // tests/planLimit.test.js (lines 53-67)

test('two simultaneous creates cannot both pass', async () => {
  await createProject(company.token, { name: 'One' });
  await createProject(company.token, { name: 'Two' });

  const [a, b] = await Promise.all([
    createProject(company.token, { name: 'Race A' }),
    createProject(company.token, { name: 'Race B' }),
  ]);

  const statuses = [a.status, b.status].sort();
  expect(statuses).toEqual([201, 402]);

  const list = await request(app).get('/api/projects').set(auth(company.token));
  expect(list.body.data.total).toBe(3);
});

    Archiving Frees a Slot

    // tests/planLimit.test.js (lines 69-84)

test('archiving a project frees a slot', async () => {
  const created = [];
  for (const name of ['One', 'Two', 'Three']) {
    created.push((await createProject(company.token, { name })).body.data.project._id);
  }
  expect((await createProject(company.token, { name: 'Blocked' })).status).toBe(402);

  await request(app).patch(`/api/projects/${created[0]}`).set(auth(company.token)).send({ isArchived: true });

  expect((await createProject(company.token, { name: 'Fits now' })).status).toBe(201);
});

    Plan Read from Database — Not JWT

    Upgrading the plan in the DB takes effect without a new token:

    // tests/planLimit.test.js (lines 150-162)

test('upgrading the database plan takes effect without a new token', async () => {
  // Token issued while tenant was on Free
  for (const name of ['One', 'Two', 'Three']) {
    expect((await createProject(company.token, { name })).status).toBe(201);
  }
  expect((await createProject(company.token, { name: 'Blocked' })).status).toBe(402);

  await setPlan(company.tenantId, 'pro'); // Direct DB update

  // Same token, no refresh: now allowed
  expect((await createProject(company.token, { name: 'Allowed after upgrade' })).status).toBe(201);
});
