    Part 4b: Tenant Plugin (tenantPlugin.js hook by hook)

    src/plugins/tenantPlugin.js

    This Mongoose plugin automatically scopes all database operations to the current tenant, preventing cross-tenant data access. It intercepts various Mongoose hooks to inject tenantId filters or validate operations.

    import mongoose from 'mongoose';
import { getTenantId, isUnscoped } from '../utils/tenantContext.js';

const READ_METHODS = [
  'find',
  'findOne',
  'findOneAndUpdate',
  'findOneAndDelete',
  'findOneAndReplace',
  'updateOne',
  'updateMany',
  'replaceOne',
  'deleteOne',
  'deleteMany',
  'countDocuments',
  'distinct',
];

const WRITE_METHODS = [
  'save',
  'create',
  'insertMany',
];

/**
 * Adds a tenantId filter to the query object, FORCE-OVERWRITING any
 * existing tenantId the caller may have passed.
 *
 * @param {Object} filter - Mongoose query filter (will be mutated)
 * @param {mongoose.Types.ObjectId} tenantId - current tenant from context
 */
function applyTenantFilter(filter, tenantId) {
  if (!filter || typeof filter !== 'object') return;

  // A filter that explicitly asks for a DIFFERENT tenant is suspicious and
  // must never be silently rewritten to "my tenant":
  //   updateMany({ tenantId: B })  ->  would become updateMany({ tenantId: A })
  // which matches every document the caller owns. Instead, make the filter
  // match nothing at all, so the operation is a no-op.
  const requested = filter.tenantId;
  if (requested !== undefined) {
    const requestedText =
      requested && typeof requested === 'object' && !requested._bsontype
        ? JSON.stringify(requested)
        : String(requested);

    if (requestedText !== String(tenantId)) {
      filter._id = { $in: [] };
    }
  }

  filter.tenantId = tenantId;
}

/**
 * Checks if an update document tries to modify tenantId.
 * Throws if $set, $unset, or direct replacement contains tenantId.
 *
 * @param {Object} update - Mongoose update document
 */
function assertNoTenantIdChange(update) {
  if (!update || typeof update !== 'object') return;

  // Direct replacement (replaceOne, findOneAndReplace with a plain object)
  if (update.tenantId !== undefined) {
    throw new Error('Changing tenantId is not allowed. A document can never move between tenants.');
  }

  // $set / $unset operators
  if (update.$set && update.$set.tenantId !== undefined) {
    throw new Error('Changing tenantId via $set is not allowed.');
  }
  if (update.$unset && update.$unset.tenantId !== undefined) {
    throw new Error('Unsetting tenantId is not allowed.');
  }
}

/**
 * Inserts a $match stage at the start of an aggregation pipeline.
 * If the pipeline starts with $geoNear (which MUST be first), we cannot
 * safely inject our $match before it, so we throw a clear error.
 *
 * @param {Array} pipeline - Aggregation pipeline array
 * @param {mongoose.Types.ObjectId} tenantId - current tenant from context
 */
function injectAggregateMatch(pipeline, tenantId) {
  if (!Array.isArray(pipeline) || pipeline.length === 0) {
    pipeline.unshift({ $match: { tenantId } });
    return;
  }

  const firstStage = pipeline[0];
  const firstStageKey = firstStage && Object.keys(firstStage)[0];

  // $geoNear MUST be the first stage in a pipeline — we cannot inject
  // a $match before it. Throw a clear error so the developer knows
  // to either run it unscoped (via runUnscoped) or restructure.
  if (firstStageKey === '$geoNear') {
    throw new Error(
      'Aggregation pipeline starts with $geoNear, which must be the first stage. ' +
      'Cannot automatically inject tenant $match. Use runUnscoped() with an explicit reason.'
    );
  }

  // Safe to inject at position 0
  pipeline.unshift({ $match: { tenantId } });
}

export function tenantPlugin(schema) {
  // ──────────────────────────────────────────────────────────────
  // a) Add tenantId field if not already present
  // ──────────────────────────────────────────────────────────────
  if (!schema.path('tenantId')) {
    schema.add({
      tenantId: {
        type: mongoose.Schema.Types.ObjectId,
        ref: 'Tenant',
        required: true,
        index: true,
      },
    });
  }

  // ──────────────────────────────────────────────────────────────
  // b) READ & WRITE queries — pre hooks to inject tenantId filter
  // ──────────────────────────────────────────────────────────────
  for (const method of READ_METHODS) {
    // Mongoose 9 uses async middleware: returning a rejected promise aborts
    // the query. There is no `next` callback anymore.
    schema.pre(method, async function () {
      // If unscoped, skip tenant filtering entirely
      if (isUnscoped()) {
        return;
      }
      // `this` is the Query object for static methods, Document for instance methods
      const tenantId = getTenantId();

      // For query middleware, `this` is the Query; filter is in this._conditions or passed as first arg
      // We need to handle both: query middleware gets filter in this.getFilter()
      if (typeof this.getFilter === 'function') {
        const filter = this.getFilter();
        applyTenantFilter(filter, tenantId);
      }
    });
  }

  // Also hook updateOne/updateMany/deleteOne/deleteMany which may receive filter as first arg
  // The above covers them since they use query middleware, but we also need to catch
  // the case where filter is passed directly. The getFilter() approach handles it.

  // ──────────────────────────────────────────────────────────────
  // c) estimatedDocumentCount — throw on scoped models
  // ──────────────────────────────────────────────────────────────
  schema.pre('estimatedDocumentCount', async function () {
    if (isUnscoped()) {
      return;
    }
    throw new Error(
      'estimatedDocumentCount() cannot be filtered by tenant and is disabled on tenant-scoped models. ' +
      'Use countDocuments() instead, or wrap in runUnscoped() with a valid reason.'
    );
  });

  // ──────────────────────────────────────────────────────────────
  // d) aggregate — inject $match as first stage (unless $geoNear)
  // ──────────────────────────────────────────────────────────────
  schema.pre('aggregate', async function () {
    if (isUnscoped()) {
      return;
    }
    const tenantId = getTenantId();
    const pipeline = this.pipeline();
    injectAggregateMatch(pipeline, tenantId);
  });

  // ──────────────────────────────────────────────────────────────
  // e) WRITES of new documents — save, create, insertMany
  // ──────────────────────────────────────────────────────────────
  // Stamp tenantId BEFORE validation. Mongoose runs `required` checks
  // during validation, which happens before pre('save'). If we only set
  // tenantId in pre('save'), a required tenantId would fail first.
  schema.pre('validate', async function () {
    if (isUnscoped() || !this.isNew) {
      return;
    }
    const tenantId = getTenantId();
    if (this.tenantId && !this.tenantId.equals(tenantId)) {
      throw new Error(
        `Document tenantId (${this.tenantId}) does not match current context (${tenantId}). ` +
        'Cannot create a document for a different tenant.'
      );
    }
    this.tenantId = tenantId;
  });

  schema.pre('save', async function () {
    if (isUnscoped()) {
      return;
    }
    const tenantId = getTenantId();

    if (this.isNew) {
      // New document — stamp tenantId
      if (this.tenantId && !this.tenantId.equals(tenantId)) {
        throw new Error(
          `Document tenantId (${this.tenantId}) does not match current context (${tenantId}). ` +
          'Cannot create a document for a different tenant.'
        );
      }
      this.tenantId = tenantId;
    } else {
      // Existing document — ensure tenantId is not being changed
      if (this.isModified('tenantId')) {
        throw new Error('Changing tenantId is not allowed. A document can never move between tenants.');
      }
    }
  });

  // create() goes through save() internally for each doc, so the pre('save') hook handles it.
  // insertMany() requires explicit tenantId in the documents passed to it.

  // ──────────────────────────────────────────────────────────────
  // f) Block tenantId changes on updates — pre hooks for update methods
  // ──────────────────────────────────────────────────────────────
  const updateMethods = ['updateOne', 'updateMany', 'findOneAndUpdate', 'replaceOne', 'findOneAndReplace'];
  for (const method of updateMethods) {
    schema.pre(method, async function () {
      if (isUnscoped()) {
        return;
      }
      // For query middleware, update document is in this._update
      const update = this.getUpdate();
      assertNoTenantIdChange(update);
    });
  }

  // ──────────────────────────────────────────────────────────────
  // g) bulkWrite — not supported on scoped models
  // ──────────────────────────────────────────────────────────────
  // bulkWrite takes an array of arbitrary operations (insertOne, updateOne,
  // deleteOne, replaceOne) each with their own filter/update documents.
  // Intercepting and rewriting every operation's filter/update safely is
  // complex and error-prone. Instead, we throw a clear error directing
  // the developer to use runUnscoped() with an explicit reason if they
  // truly need bulkWrite (rare — usually for migrations or seeding).
  schema.pre('bulkWrite', async function () {
    if (isUnscoped()) {
      return;
    }
    throw new Error(
      'bulkWrite() is not supported on tenant-scoped models. ' +
      'It accepts arbitrary operations that cannot be safely auto-scoped. ' +
      'Use runUnscoped(reason, callback) with a valid justification if absolutely necessary.'
    );
  });
}

export default tenantPlugin;

    Explanation of tenantPlugin.js (hook by hook):

    a) Add tenantId field

    • Code: if (!schema.path('tenantId')) { schema.add({ tenantId: { type: mongoose.Schema.Types.ObjectId, ref: 'Tenant', required: true, index: true, }, }); }
    • Explanation: This ensures that every schema where the tenantPlugin is applied automatically gets a tenantId field. It's a required field, indexed for performance, and references the Tenant model.

    b) Query Methods (find, findOne, updateOne, deleteOne, etc.)

    • Code: for (const method of READ_METHODS) { schema.pre(method, async function () { ... }); }
    • Explanation: For every read and write method listed in READ_METHODS, a pre hook is registered. This hook runs before the actual database operation.
    • getFilter(): Inside the hook, this.getFilter() retrieves the current query filter.
    • Force-overwrite caller's tenantId: The applyTenantFilter function is called, which gets the tenantId from the current context using getTenantId(). It then force-sets the tenantId in the query filter to ensure all operations are scoped to the current tenant.
    • Mismatch matching empty array { _id: { $in: [] } }: If the incoming query explicitly specified a different tenantId than the one in the current context, instead of overwriting it (which could lead to widening the query to all documents of the current tenant), the filter is modified to match nothing (_id: { $in: [] }). This makes the operation a safe no-op.
    • What happens when no tenant is in context: If getTenantId() is called when no tenant is in context (i.e., outside of runWithTenant and not within runUnscoped), it throws an error. This is a critical safety measure to prevent accidental data leaks across tenants by making such an oversight loud and immediate.

    c) estimatedDocumentCount

    • Code: schema.pre('estimatedDocumentCount', async function () { ... });
    • Explanation: This hook explicitly throws an error if estimatedDocumentCount() is called on a tenant-scoped model outside of an runUnscoped block. estimatedDocumentCount cannot be filtered by design, so allowing it would break tenant isolation. Developers are directed to use countDocuments() (which is tenant-scoped) or explicitly use runUnscoped() if they truly need an unscoped count.

    d) Aggregation

    • Code: schema.pre('aggregate', async function () { ... });
    • Explanation: For aggregation pipelines, a pre('aggregate') hook is used to inject a $match stage at the very beginning of the pipeline. This $match stage filters documents by the current tenantId.
    • Throwing on $geoNear: If the pipeline starts with a $geoNear stage, the plugin throws an error. This is because $geoNear must be the first stage in an aggregation pipeline, and injecting $match before it would break its functionality. In such cases, developers must explicitly use runUnscoped().
    • $lookup and $unionWith limitations: The documentation explicitly notes that $lookup and $unionWith operations, which join data from other collections, are not automatically scoped on their joined target collections. Developers must manually add $match stages within these operations to ensure tenant isolation for the joined data.

    e) Writes of new documents (save, create, insertMany)

    • pre('validate') hook:
      • Code: schema.pre('validate', async function () { ... });
      • Explanation: This hook runs before document validation. If a new document (this.isNew) is being created and it already has a tenantId that doesn't match the current context, it throws an error. Otherwise, it stamps the tenantId from the current context onto the new document. This is done here to ensure tenantId is present before Mongoose's `required` validation runs.
    • pre('save') hook:
      • Code: schema.pre('save', async function () { ... });
      • Explanation: For new documents, it re-verifies and stamps the tenantId if not already present, ensuring consistency. For existing documents, it checks if the tenantId field is being modified (this.isModified('tenantId')). If so, it throws an error, as a document can never change its tenant.

    f) Block tenantId changes on updates

    • Code: const updateMethods = [...]; for (const method of updateMethods) { schema.pre(method, async function () { ... }); }
    • Explanation: For update operations (updateOne, updateMany, findOneAndUpdate, replaceOne, findOneAndReplace), a pre hook intercepts the update document. The assertNoTenantIdChange function is called to verify that the update operation is not attempting to modify ($set, $unset, or direct replacement) the tenantId field. This is crucial for maintaining data integrity and tenant isolation.

    g) bulkWrite

    • Code: schema.pre('bulkWrite', async function () { ... });
    • Explanation: The bulkWrite() method allows for a complex array of operations, making safe automatic tenant scoping extremely difficult and error-prone. Therefore, this hook simply throws an error if bulkWrite() is called on a tenant-scoped model outside of an runUnscoped block. Developers are instructed to use runUnscoped() with a clear justification if they absolutely need to use bulkWrite (typically for system-level tasks).
