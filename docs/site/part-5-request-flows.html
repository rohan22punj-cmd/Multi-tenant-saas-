    Step 2: protect.js — JWT Verification & Context Setup

    // src/middleware/protect.js (lines 29-59)

const protect = async (req, _res, next) => {
  // 1. Extract token from Authorization header
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    throw new AppError('Unauthorized', 401);
  }
  const token = authHeader.split(' ')[1];

  // 2. Verify signature + expiry using ACCESS secret
  let decoded;
  try {
    decoded = verifyAccessToken(token);  // throws if invalid/expired
  } catch {
    throw new AppError('Unauthorized', 401);
  }

  // 3. Build req.user from token payload (NO database call)
  req.user = {
    userId: decoded.userId,
    tenantId: decoded.tenantId,
    roleId: decoded.roleId,
    permissions: decoded.permissions || [],
  };

  // 4. Run downstream inside async tenant context
  runWithTenant(
    { tenantId: decoded.tenantId, userId: decoded.userId },
    () => next()
  );
};

    Key point: req.user.permissions comes directly from the JWT — no DB lookup. The viewer's token has ["project:read", "task:create", "task:read", "task:update"].

    Step 3: AsyncLocalStorage Tenant Context

    // src/utils/tenantContext.js (lines 40-62)

export function runWithTenant(context, callback) {
  return asyncLocalStorage.run(context, callback);
}

export function getTenantId() {
  if (unscopedDepth > 0) return undefined;
  const store = asyncLocalStorage.getStore();
  if (!store?.tenantId) {
    throw new Error('getTenantId() called outside a tenant context.');
  }
  return store.tenantId;
}

    The runWithTenant call in protect.js stores { tenantId: "tenant-A", userId: "viewer-123" } in AsyncLocalStorage. All downstream code (controllers, services, Mongoose plugins) can call getTenantId() to read it without passing it through function arguments.

    Step 4: requirePermission Middleware — Checks Permissions

    // src/middleware/requirePermission.js (lines 26-37)

const requirePermission = (...keys) => {
  validateKeys(keys);  // throws at startup if unknown key
  return (req, _res, next) => {
    if (!req.user) return next(new AppError('Unauthorized', 401));
    const missing = keys.filter(key => !req.user.permissions.includes(key));
    if (missing.length > 0) {
      return next(new AppError('Forbidden', 403));  // ← VIEWER HITS THIS
    }
    next();
  };
};

    The route is defined as:

    // src/routes/projectRoutes.js
router.delete('/:id', protect, requirePermission('project:delete'), projectController.remove);

    The viewer's permissions array does not include project:delete. The middleware returns 403 immediately — the controller never runs.

    Step 6: tenantPlugin Auto-Injects Company B's tenantId

    // src/services/projectService.js (line 93-98)

export async function getProject({ id }) {
  const project = await Project.findById(id);  // ← tenantPlugin intercepts this
  if (!project) throw new AppError('Project not found', 404);
  return project;
}

    The tenantPlugin pre-hook on findById runs:

    // src/plugins/tenantPlugin.js (lines 166-183)

for (const method of READ_METHODS) {
  schema.pre(method, async function () {
    if (isUnscoped()) return;
    const tenantId = getTenantId();  // Returns "tenant-B"
    if (typeof this.getFilter === 'function') {
      const filter = this.getFilter();
      applyTenantFilter(filter, tenantId);  // Forces filter.tenantId = "tenant-B"
    }
  });
}

    Even though the caller passed findById("project-A-id") (Company A's project), the plugin rewrites the filter to { _id: "project-A-id", tenantId: "tenant-B" }.

    Step 7: findById Returns null → 404

    // applyTenantFilter (src/plugins/tenantPlugin.js lines 71-92)

function applyTenantFilter(filter, tenantId) {
  // Caller asked for project-A-id with NO tenantId in filter
  // Plugin force-sets filter.tenantId = "tenant-B"
  filter.tenantId = tenantId;
  // Query becomes: { _id: "project-A-id", tenantId: "tenant-B" }
  // No document matches both → returns null
}
    HTTP/1.1 404 Not Found
Content-Type: application/json

{
  "message": "Project not found"
}

    Company B never sees Company A's data. The tenant plugin guarantees isolation at the query level.

    Stripe signature verification requires the exact raw body. Standard express.json() would parse it and break verification.

    Step 2: Signature Verification

    // PLANNED — not yet implemented

const sig = req.headers['stripe-signature'];
let event;

try {
  event = stripe.webhooks.constructEvent(
    req.body,                      // raw Buffer
    sig,
    process.env.STRIPE_WEBHOOK_SECRET
  );
} catch (err) {
  return res.status(400).send(`Webhook Error: ${err.message}`);
}

    If signature verification fails, we return 400 immediately. Stripe will retry.

    Step 3: Redis Atomic Claim (SET NX EX)

    // PLANNED — not yet implemented

const eventId = event.id;
const claimed = await redis.set(
  `stripe:event:${eventId}`,
  'processing',
  'NX',      // Only set if key does NOT exist
  'EX', 86400  // 24-hour TTL
);

if (!claimed) {
  // Another process already claimed this event — acknowledge to Stripe
  return res.json({ received: true });
}

    SET key NX EX is atomic. If two webhook deliveries arrive simultaneously, only one gets claimed === true. The other gets claimed === false and returns 200 immediately (acknowledging to Stripe without reprocessing).

    Step 4: Idempotency Check — Process or Release

    // PLANNED — not yet implemented

try {
  await handleStripeEvent(event);  // Business logic: update plan, etc.
  await redis.set(`stripe:event:${eventId}`, 'done', 'EX', 86400);
  res.json({ received: true });
} catch (err) {
  // On error, release the claim so a retry can pick it up
  await redis.del(`stripe:event:${eventId}`);
  throw err;  // Let Express error handler return 500 → Stripe retries
}

    If processing succeeds, mark as done. If it fails, delete the key so the next delivery can retry.

    Step 5: Fallback — MongoDB Unique Index (if Redis unavailable)

    // PLANNED — not yet implemented

const stripeEventSchema = new mongoose.Schema({
  eventId: { type: String, required: true, unique: true },  // ← unique index
  type: String,
  payload: mongoose.Schema.Types.Mixed,
  status: { type: String, enum: ['pending', 'done', 'failed'], default: 'pending' },
  createdAt: { type: Date, default: Date.now },
});

export const StripeEvent = mongoose.model('StripeEvent', stripeEventSchema);

// In handler (fallback path):
try {
  await StripeEvent.create({ eventId, type: event.type, payload: event.data.object });
  await handleStripeEvent(event);
  await StripeEvent.updateOne({ eventId }, { status: 'done' });
} catch (err) {
  if (err.code === 11000) {
    // Duplicate key — already processed
    return res.json({ received: true });
  }
  await StripeEvent.updateOne({ eventId }, { status: 'failed' });
  throw err;
}

    The unique index on eventId provides the same idempotency guarantee with MongoDB durability. Used if Redis is down or as a secondary check.
