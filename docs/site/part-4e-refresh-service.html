    Part 4e: Refresh Service & Token Rotation (src/services/authService.js)

    src/services/authService.js

    The refresh() function handles token renewal with rotation and reuse detection. The logout() function clears the refresh token hash. Both use separate access and refresh secrets for security.

    export async function refresh(refreshToken) {
  const GENERIC_MSG = 'Invalid refresh token';

  // 1. Verify signature and expiry
  let decoded;
  try {
    decoded = verifyRefreshToken(refreshToken);
  } catch {
    throw new AppError(GENERIC_MSG, 401);
  }

  // 2. Find the user and check tenant INSIDE tenant context
  const { userId, tenantId } = decoded;

  const user = await runWithTenant({ tenantId }, async () => {
    return User.findById(userId);
  });
  if (!user || !user.isActive) {
    throw new AppError(GENERIC_MSG, 401);
  }

  // 3. Check that the tenant still exists (unscoped - Tenant model)
  const tenant = await runUnscoped(
    'refresh: verify tenant still exists',
    () => Tenant.findById(tenantId)
  );
  if (!tenant) {
    throw new AppError(GENERIC_MSG, 401);
  }

  // 4. Hash the incoming token and compare with stored hash
  const incomingHash = hashToken(refreshToken);

  if (user.refreshTokenHash !== incomingHash) {
    // REUSE DETECTED — wipe the stored hash to kill all sessions
    await runWithTenant({ tenantId }, async () => {
      user.refreshTokenHash = null;
      await user.save();
    });
    throw new AppError(GENERIC_MSG, 401);
  }

  // 5. Fetch the role to get current permissions
  const role = await runWithTenant({ tenantId }, async () => {
    return Role.findById(user.roleId).lean();
  });
  if (!role) {
    throw new AppError(GENERIC_MSG, 401);
  }

  // 6. Issue fresh tokens (this also replaces the stored hash)
  const tokens = await runWithTenant({ tenantId }, async () => {
    return issueAndStoreTokens(user, role.permissions);
  });

  return tokens;
}

export async function logout(refreshToken) {
  // Best-effort: try to decode the token to find the user.
  // If it's garbage or expired, we just do nothing — still return success.
  try {
    const decoded = verifyRefreshToken(refreshToken);
    const { userId, tenantId } = decoded;

    await runWithTenant({ tenantId }, async () => {
      await User.findByIdAndUpdate(userId, { refreshTokenHash: null });
    });
  } catch {
    // Token is garbage or expired — nothing to clear, that's fine.
  }
}

    Explanation:

    • Refresh token verification (refreshSecret): The verifyRefreshToken function checks the signature using the refreshSecret and the token's expiry.
    • Token hashing (SHA-256): The incoming refresh token is hashed using hashToken (SHA-256) and compared to the stored hash.
    • Token rotation & reuse detection: If the hash matches, new tokens are issued and the stored hash is replaced (rotation). If the hash does NOT match, it means an old token is being reused — the stored hash is cleared to null, which kills all user sessions.
    • Separate secrets: Access tokens and refresh tokens use different secrets (ACCESS_SECRET and REFRESH_SECRET) defined in config/env.js.
