    Webhook Endpoint Design

    Stripe requires the raw request body for signature verification. Standard express.json() middleware parses the body, which breaks signature verification. The endpoint must use express.raw({ type: 'application/json' }) instead.

    Signature Verification

    // Planned — not yet implemented

app.post(
  '/api/webhooks/stripe',
  express.raw({ type: 'application/json' }),
  async (req, res) => {
    const sig = req.headers['stripe-signature'];
    let event;

    try {
      event = stripe.webhooks.constructEvent(
        req.body,                    // raw Buffer, NOT parsed JSON
        sig,
        process.env.STRIPE_WEBHOOK_SECRET
      );
    } catch (err) {
      return res.status(400).send(`Webhook Error: ${err.message}`);
    }

    // Handle event types...
  }
);

    Idempotency Design

    Stripe may deliver the same event multiple times. We must process each event exactly once.

    Primary: Redis Claim Pattern

    On receipt, atomically claim the event ID using SET key NX EX. If the claim fails, the event was already processed.

    // Planned — not yet implemented

const eventId = event.id;
const claimed = await redis.set(
  `stripe:event:${eventId}`,
  'processing',
  'NX',    // only set if not exists
  'EX', 86400 // 24h TTL
);

if (!claimed) {
  // Already processed (or being processed) — acknowledge to Stripe
  return res.json({ received: true });
}

try {
  await handleStripeEvent(event);
  await redis.set(`stripe:event:${eventId}`, 'done', 'EX', 86400);
  res.json({ received: true });
} catch (err) {
  // Release claim on error so a retry can pick it up
  await redis.del(`stripe:event:${eventId}`);
  throw err;
}

    Fallback: StripeEvent Collection

    If Redis is unavailable, a MongoDB collection with a unique index on eventId provides durability:

    // Planned — not yet implemented

import mongoose from 'mongoose';

const stripeEventSchema = new mongoose.Schema({
  eventId: { type: String, required: true, unique: true },
  type: String,
  payload: mongoose.Schema.Types.Mixed,
  status: { type: String, enum: ['pending', 'done', 'failed'], default: 'pending' },
  createdAt: { type: Date, default: Date.now },
});

export const StripeEvent = mongoose.model('StripeEvent', stripeEventSchema);

// In handler:
try {
  await StripeEvent.create({ eventId, type: event.type, payload: event.data.object });
  await handleStripeEvent(event);
  await StripeEvent.updateOne({ eventId }, { status: 'done' });
} catch (err) {
  if (err.code === 11000) {
    // Duplicate key — already processed
    return res.json({ received: true });
  }
  await StripeEvent.updateOne({ eventId }, { status: 'failed' });
  throw err;
}
